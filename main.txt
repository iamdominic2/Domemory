Define Joinstring B(string nw) (warp=false)
{
    JOINSTRING STRING = "";
    JOINSTRING NO = 0;
    Repeat ((Variable.Length(TextEdit) - 1))
    {
        JOINSTRING NO += 1;
        JOINSTRING STRING = Operator.Join(JOINSTRING STRING, Operator.LetterOf(TextEdit, JOINSTRING NO));
    }
    JOINSTRING STRING = Operator.Join(JOINSTRING STRING, nw);
}

Define Joinstring text(string st) Location(string ct) Increment(string inc) new(string nw) (warp=false)
{
    JOINSTRING STRING = "";
    JOINSTRING NO = 0;
    Repeat (ct)
    {
        JOINSTRING NO += 1;
        JOINSTRING STRING = Operator.Join(JOINSTRING STRING, Operator.LetterOf(st, JOINSTRING NO));
    }
    JOINSTRING STRING = Operator.Join(JOINSTRING STRING, nw);
    JOINSTRING NO += inc;
    Repeat (((Variable.Length(st) - ct) - inc))
    {
        JOINSTRING NO += 1;
        JOINSTRING STRING = Operator.Join(JOINSTRING STRING, Operator.LetterOf(st, JOINSTRING NO));
    }
    cursor location += (0 - (inc + back));
}

Define Key(string s) (warp=false)
{
    If (Variable.Length(TextEdit) < 200)
    {
        If (Not (Sensing.KeyPressed(s)))
        {
            List.DeleteItem(Keys, List.IndexOf(Keys, s));
        }
        If (Sensing.KeyPressed(s) And (Not (List.ContainsItem(Keys, s))))
        {
            Sensing.ResetTimer();
            If (s == "enter")
            {
                Call Joinstring text(TextEdit) Location((cursor location - 1)) Increment("") new("☙");
            }
            Else
            {
                If ((INSMODE == 1) And (cursor location < Variable.Length(TextEdit)))
                {
                    Call Joinstring text(TextEdit) Location((cursor location - 1)) Increment(1) new(s);
                    cursor location += 1;
                }
                Else
                {
                    If (cursor location == Variable.Length(TextEdit))
                    {
                        Call Joinstring B(s);
                    }
                    Else
                    {
                        Call Joinstring text(TextEdit) Location((cursor location - 1)) Increment("") new(s);
                    }
                }
            }
            Call Typer color;
            List.Add(Keys, s);
            cursor location += 1;
            TextEdit = JOINSTRING STRING;
        }
    }
}
